# AI E2E Job Portal — pinned dependencies (verified on Python 3.12.10)
#
#   python -m venv .venv
#   .venv\Scripts\python -m pip install -r requirements.txt     # Windows
#   .venv/bin/python  -m pip install -r requirements.txt        # Linux / macOS
#
# Every entry is free and open source. There is no cloud SDK in here on purpose: the
# Gemini backend is plain REST over `requests`, the same shape as the Ollama and
# OpenAI-compatible backends, so moving to a local model is an env var and not an
# install. Versions are pinned to what the build was actually verified against.
#
# Torch is only used by ultralytics (proctoring object detection) and this project
# never needs a GPU. The CPU index below serves 2.14.0+cpu, which outranks the plain
# 2.14.0 on PyPI, so `pip install -r` picks the small CPU wheel rather than the
# multi-gigabyte CUDA one.
--extra-index-url https://download.pytorch.org/whl/cpu
